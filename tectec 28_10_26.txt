{

	{
		inteiro x, n, primo
		escreva ("--- Digite um Numero: ---\n")
		leia(n)
		
		x = 1
		primo = 1
		
		escreva (n+ " é divisivel por ")

    	escreva ("\nO número "+n+" não é primo,\ne foram encontrados "+primo+" divisores")
    }
	}
